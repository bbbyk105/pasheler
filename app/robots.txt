// app/robots.ts
import type { MetadataRoute } from "next";

export default function robots(): MetadataRoute.Robots {
  return {
    rules: [
      {
        userAgent: "*",
        allow: "/", // 通常ページはクロールOK
        disallow: ["/api/", "/_next/", "/private/"], // APIや内部構造は拒否
      },
    ],
    sitemap: "https://yawnnap.shop/sitemap.xml",
    host: "https://yawnnap.shop",
  };
}
